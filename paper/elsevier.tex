% CAS captures the abstract environment verbatim in \jobname.abs.  An
% \input command in that environment is therefore not a reliable shared-source
% adapter: it can survive as literal payload rather than supplying the prose.
% Copy abstract.tex line-for-line into the CAS auxiliary file instead.  The
% wrapper remains mechanical and the abstract continues to have one source.
\newread\casabstractsource
\newwrite\casabstractcapture
\newcommand{\capturecasabstract}[1]{%
  \openin\casabstractsource=#1\relax
  \ifeof\casabstractsource
    \PackageError{concentration-is-not-scaling}{Cannot open shared abstract #1}{}%
  \else
    \immediate\openout\casabstractcapture=\jobname.abs\relax
    \begingroup
      \endlinechar=-1
      \loop\unless\ifeof\casabstractsource
        \readline\casabstractsource to \casabstractline
        \immediate\write\casabstractcapture{\casabstractline}%
      \repeat
    \endgroup
    \immediate\closeout\casabstractcapture
  \fi
  \closein\casabstractsource
}

\begin{highlights}
\item Concentration does not identify aggregate-variance scaling.
\item Variance elasticities admit an exact finite-interval decomposition.
\item Signed covariance enters without logarithms of component values.
\item Equal concentration paths can imply different diversification rates.
\item Positive covariance can contribute negatively to variance decay.
\end{highlights}
